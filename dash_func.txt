        function updateDashboard() {
            // حساب المبيعات من جميع الفواتير المُسلّمة
            const deliveredInvoices = invoices.filter(invoice => invoice.deliveryStatus === 'delivered');
            
            // إجمالي المبيعات = مجموع سعر البيع بدون خصم أي شيء
            const totalRevenue = deliveredInvoices.reduce((sum, invoice) => {
                const items = normalizeInvoiceItems(invoice);
                return sum + items.reduce((itemSum, item) => itemSum + (item.unitPrice || 0) * (item.quantity || 0), 0);
            }, 0);
            
            // Count all invoices as orders
            const totalOrders = invoices.length;
            const totalProductsCount = products.length;
            // Only count confirmed expenses in total expenses
            const confirmedExpenses = expenses.filter(e => e.confirmed);
            const totalExpensesAmount = confirmedExpenses.reduce((sum, expense) => sum + expense.amount, 0);
            
            // Show ALL expenses in dashboard (not just confirmed ones)
            const totalExpensesDisplay = expenses.reduce((sum, expense) => sum + expense.amount, 0);
            
            // إجمالي الربح = المبيعات - العمولة - الشحن - الضريبة
            let grossProfit = 0;
            deliveredInvoices.forEach(invoice => {
                const items = normalizeInvoiceItems(invoice);
                items.forEach(item => {
                    const product = products.find(p => p.id == item.productId);
                    if (!product) return;
                    
                    const unitPrice = item.unitPrice || 0;
                    const quantity = item.quantity || 0;
                    const sellingAmount = unitPrice * quantity;
                    
                    // خصم العمولة والشحن والضريبة
                    const commission = sellingAmount * (invoice.commissionPercent || 0) / 100;
                    const shipping = (invoice.shipping || 0);
                    const tax = sellingAmount * 0.14; // ضريبة 14%
                    
                    // الربح = سعر البيع - العمولة - الشحن - الضريبة
                    grossProfit += (sellingAmount - commission - shipping - tax);
                });
            });
            
            // استخراج خسائر المرتجعات
            const totalReturnLosses = returns.reduce((sum, ret) => {
                return sum + (ret.totalLoss || 0);
            }, 0);
            
            // صافي الربح = إجمالي الربح - المصروفات - خسائر المرتجعات
            const netProfit = grossProfit - totalReturnLosses - totalExpensesAmount;
            
            // تحديث الواجهة
            document.getElementById('totalRevenue').textContent = totalRevenue.toFixed(2) + ' جنيه';
            document.getElementById('totalOrders').textContent = totalOrders;
            document.getElementById('totalProducts').textContent = totalProductsCount;
            document.getElementById('totalExpenses').textContent = totalExpensesDisplay.toFixed(2) + ' جنيه';
            document.getElementById('grossProfit').textContent = grossProfit.toFixed(2) + ' جنيه';
            
            // تحديث صافي الربح مع اللون
            const netProfitElement = document.getElementById('netProfit');
            netProfitElement.textContent = netProfit.toFixed(2) + ' جنيه';

            let totalUnitsSoldDash = 0;
            deliveredInvoices.forEach(inv => {
                const items = normalizeInvoiceItems(inv);
                items.forEach(it => { totalUnitsSoldDash += (it.quantity || 0); });
            });
            const avgProfitVal = totalUnitsSoldDash > 0 ? (netProfit / totalUnitsSoldDash) : 0;
            const avgElem = document.getElementById('avgProfitPerUnit');
            if (avgElem) avgElem.textContent = avgProfitVal.toFixed(2) + ' جنيه';
            netProfitElement.style.color = netProfit >= 0 ? '#28a745' : '#dc3545';
            
            // حساب الزكاة 10% من صافي الربح
            const positiveNetProfit = netProfit > 0 ? netProfit : 0;
            const ZainetProfit = positiveNetProfit * 0.10;
            const netAfterZakat = positiveNetProfit - ZainetProfit;
            
            document.getElementById('zakatAmount').textContent = ZainetProfit.toFixed(2) + ' جنيه';
            document.getElementById('netAfterZakat').textContent = netAfterZakat.toFixed(2) + ' جنيه';
            
            // Calculate profit percentage
            const profitPercentage = totalRevenue > 0 ? ((netProfit / totalRevenue) * 100).toFixed(1) : 0;
            document.getElementById('profitPercentage').textContent = profitPercentage + '%';
            
            // تحديث لون نسبة الربح حسب النتيجة
            const profitPercentageElement = document.getElementById('profitPercentage');
            profitPercentageElement.parentElement.querySelector('small').innerHTML = netProfit >= 0 ? '<i class="fas fa-arrow-up"></i> ممتاز' : '<i class="fas fa-arrow-down"></i> خسارة';
            
            // تحديث رأس المال في الداشبورد
            if (document.getElementById('dashboardCapital')) {
                const currentCapital = capitalAmount || 0;
                document.getElementById('dashboardCapital').textContent = currentCapital.toFixed(2) + ' جنيه';
            }
