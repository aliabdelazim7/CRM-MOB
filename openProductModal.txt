        
        function openProductModal(productId = null) {
            console.log('openProductModal called with productId:', productId);
            // Load collection and supplier options first
            loadCollectionOptions();
            loadProductSupplierOptions();
            
            if (productId) {
                editingProductId = productId;
                const product = products.find(p => p.id === productId);
                if (product) {
                    document.getElementById('productName').value = product.name;
                    document.getElementById('productCode').value = product.code;
                    document.getElementById('productBarcode').value = product.barcode || '';
                    document.getElementById('productCollection').value = product.collection;
                    document.getElementById('productSupplier').value = product.supplier;
                    document.getElementById('purchasePrice').value = product.purchasePrice || 0;
                    document.getElementById('sellingPrice').value = product.sellingPrice;
                    document.getElementById('minStock').value = product.minStock || 5;
                    
                    // Handle noon price - use original price if available, otherwise calculate from VAT price
                    let originalNoonPrice = product.noonPrice || '';
                    if (!product.noonPriceWithVAT && product.noonPrice > 0) {
                        // Old product - calculate original price from VAT price
                        originalNoonPrice = (product.noonPrice / 1.14).toFixed(2);
                    }
                    document.getElementById('noonPrice').value = originalNoonPrice;
                    document.getElementById('noonCommission').value = product.noonCommission || '';
                    document.getElementById('noonShipping').value = product.noonShipping || '';
                    document.getElementById('noonAdsExpense').value = product.noonAds || 0;
                    document.getElementById('jumiaPrice').value = product.jumiaPrice || '';
                    document.getElementById('jumiaCommission').value = product.jumiaCommission || '';
                    document.getElementById('jumiaShipping').value = product.jumiaShipping || '';
                    document.getElementById('jumiaAdsExpense').value = product.jumiaAds || 0;
                    document.getElementById('amazonPrice').value = product.amazonPrice || '';
                    document.getElementById('amazonCommission').value = product.amazonCommission || '';
                    document.getElementById('amazonAdsExpense').value = product.amazonAds || 0;
                    const otherPriceInput = document.getElementById('otherPrice');
                    if(otherPriceInput) otherPriceInput.value = product.otherPrice || '';
                    document.getElementById('siteAdsExpense').value = product.siteAds || 0;

                    // Clear and load custom stores
                    const customContainer = document.getElementById('customStoresContainer');
                    if (customContainer) customContainer.innerHTML = '';
                    if (product.customStores && Array.isArray(product.customStores)) {
                        product.customStores.forEach(st => addCustomStoreField(st));
                    }
                    
                    const imgUrlInput = document.getElementById('productImageUrl');
                    if (imgUrlInput) imgUrlInput.value = product.imageUrl || (typeof product.image === 'string' && product.image.startsWith('http') ? product.image : '');
                    
                    const imgPreviewCont = document.getElementById('productImagePreviewContainer');
                    const imgPreview = document.getElementById('productImagePreview');
                    if (imgPreviewCont && imgPreview && product.image) {
                        imgPreview.src = product.image;
                        imgPreviewCont.style.display = 'block';
                    } else if (imgPreviewCont) {
                        imgPreviewCont.style.display = 'none';
                    }
                    
                    // Load colors
                    const colorsContainer = document.getElementById('colorsContainer');
                    colorsContainer.innerHTML = ''; // Clear existing colors
                    
                    if (product.colors && product.colors.length > 0) {
                        // New format with multiple colors
                        product.colors.forEach(color => {
                            const colorItem = document.createElement('div');
                            colorItem.className = 'color-item';
                            colorItem.style.cssText = 'display: flex; gap: 10px; margin-bottom: 10px; align-items: center; padding: 15px; background: #f8f9fa; border-radius: 8px; flex-wrap: wrap;';
                            const barcodeValue = color.barcode || 'C' + Date.now() + Math.floor(Math.random() * 1000);
                            colorItem.innerHTML = `
                                <input type="text" placeholder="اللون" class="color-name" value="${color.name}" style="flex: 1; min-width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                                <input type="text" placeholder="كود اللون" class="color-code" value="${color.code || ''}" style="width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                                <input type="text" placeholder="باركود اللون" class="color-barcode" value="${barcodeValue}" style="width: 130px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                                <input type="number" placeholder="العدد" class="color-quantity" value="${color.quantity}" min="0" style="width: 80px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                                <button type="button" onclick="removeColorItem(this)" class="btn btn-delete" style="padding: 8px;">
                                    <i class="fas fa-trash"></i>
                                </button>
                            `;
                            colorsContainer.appendChild(colorItem);
                        });
                    } else {
                        // Old format - convert to new format
                        const colorItem = document.createElement('div');
                        colorItem.className = 'color-item';
                        colorItem.style.cssText = 'display: flex; gap: 10px; margin-bottom: 10px; align-items: center; padding: 15px; background: #f8f9fa; border-radius: 8px; flex-wrap: wrap;';
                        const oldBarcode = product.barcode || product.colorCode || ('C' + Date.now() + Math.floor(Math.random() * 1000));
                        colorItem.innerHTML = `
                            <input type="text" placeholder="اللون" class="color-name" value="${product.color || ''}" style="flex: 1; min-width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                            <input type="text" placeholder="كود اللون" class="color-code" value="${product.colorCode || ''}" style="width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                            <input type="text" placeholder="باركود اللون" class="color-barcode" value="${oldBarcode}" style="width: 130px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                            <input type="number" placeholder="العدد" class="color-quantity" value="${product.quantity || 0}" min="0" style="width: 80px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                            <button type="button" onclick="removeColorItem(this)" class="btn btn-delete" style="padding: 8px;">
                                <i class="fas fa-trash"></i>
                            </button>
                        `;
                        colorsContainer.appendChild(colorItem);
                    }
                    
                    // Load platform discounts
                    const discountsList = document.getElementById('platformDiscountsList');
                    discountsList.innerHTML = '';
                    if (product.platformDiscounts && product.platformDiscounts.length > 0) {
                        product.platformDiscounts.forEach(discount => {
                            const discountId = Date.now() + Math.random();
                            const discountItem = document.createElement('div');
                            discountItem.className = 'platform-discount-item';
                            discountItem.id = 'discount_' + discountId;
                            discountItem.style.cssText = 'display: flex; gap: 8px; align-items: center; padding: 8px; background: #f8f9fa; border-radius: 6px; margin-bottom: 8px; flex-wrap: wrap;';
                            discountItem.innerHTML = `
                                <select id="discountPlatform_${discountId}" onchange="calculatePlatformProfits()" style="padding: 6px; border: 1px solid #ddd; border-radius: 4px; flex: 1; min-width: 100px; font-size: 0.85rem;">
                                    <option value="website" ${discount.platform === 'website' ? 'selected' : ''}>موقع الويب</option>
                                    <option value="jumia" ${discount.platform === 'jumia' ? 'selected' : ''}>جوميا</option>
                                    <option value="noon" ${discount.platform === 'noon' ? 'selected' : ''}>نون</option>
                                </select>
                                <input type="number" id="discountValue_${discountId}" oninput="calculatePlatformProfits()" value="${discount.price}" placeholder="السعر" step="0.01" style="padding: 6px; border: 1px solid #ddd; border-radius: 4px; width: 100px; font-size: 0.85rem;">
                                <span style="color: #666; font-size: 0.8rem;">جنيه</span>
                                <button type="button" onclick="removePlatformDiscount(${discountId})" class="btn btn-delete" style="padding: 5px 8px; font-size: 0.8rem;">
                                    <i class="fas fa-trash"></i>
                                </button>
                            `;
                            discountsList.appendChild(discountItem);
                        });
                    }
                    
                    document.querySelector('#productModal .modal-header h3').innerHTML = '<i class="fas fa-edit"></i> تعديل المنتج';
                    calculateProfits(); // Calculate profits after loading data
                }
            } else {
                // Reset form for new product
                document.getElementById('productForm').reset();
                // Auto-generate unique barcode for new product
                const newBarcode = 'HC' + Date.now().toString().slice(-6) + Math.floor(Math.random() * 1000).toString().padStart(3, '0');
                document.getElementById('productCode').value = newBarcode;
                document.getElementById('productBarcode').value = newBarcode;
                
                // Clear platform discounts list for new product
                document.getElementById('platformDiscountsList').innerHTML = '';
                calculatePlatformProfits();
                
                const colorsContainer = document.getElementById('colorsContainer');
                const defaultBarcode = 'C' + Date.now() + Math.floor(Math.random() * 1000);
                // Also set the main barcode field to match
                document.getElementById('productBarcode').value = defaultBarcode;
                colorsContainer.innerHTML = `
                    <div class="color-item" style="display: flex; gap: 10px; margin-bottom: 10px; align-items: center; padding: 15px; background: #f8f9fa; border-radius: 8px; flex-wrap: wrap;">
                        <input type="text" placeholder="اللون" class="color-name" style="flex: 1; min-width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                        <input type="text" placeholder="كود اللون" class="color-code" style="width: 100px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                        <input type="text" placeholder="باركود اللون" class="color-barcode" value="${defaultBarcode}" style="width: 130px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                        <input type="number" placeholder="العدد" class="color-quantity" min="0" style="width: 80px; padding: 8px; border: 1px solid #ddd; border-radius: 5px;">
                        <button type="button" onclick="removeColorItem(this)" class="btn btn-delete" style="padding: 8px;">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
                `;
                document.querySelector('#productModal .modal-header h3').innerHTML = '<i class="fas fa-box"></i> إضافة منتج جديد';
                
                // Initialize profit displays for new product
                calculateProfits();
