                supplierOrders[supplier].push(product);
            });
            
            // Create purchase orders for each supplier
            Object.keys(supplierOrders).forEach(supplier => {
                const products = supplierOrders[supplier];
                products.forEach(product => {
                    const orderQuantity = (product.minStock || 5) - product.totalQuantity + 10; // Order enough to reach min + 10 extra
                    
                    const newPurchase = {
                        id: Date.now() + Math.floor(Math.random() * 1000),
                        purchaseSupplier: supplier,
                        purchaseSupplierAddress: '',
                        productId: product.id,
                        purchasePrice: product.purchasePrice || 0,
                        purchaseQuantity: orderQuantity,
                        purchaseDate: new Date().toISOString().split('T')[0],
                        purchaseNotes: 'طلب شراء تلقائي للمخزون المنخفض',
                        status: 'pending',
                        autoGenerated: true,
                        createdAt: new Date().toISOString()
                    };
                    
                    purchases.push(newPurchase);
                });
